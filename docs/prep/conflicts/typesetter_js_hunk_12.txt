    if (finalDiff > 0.5) {

    const overshootBeforeShrink = targetCol ? targetCol.scrollHeight - targetHeight : null;
    if (targetCol && overshootBeforeShrink > 0.5) {

    const rollbackReason = best.fits && justifyOverflows
